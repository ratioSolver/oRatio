cmake_minimum_required(VERSION 3.21)
project(oRatioLib VERSION 0.3.0 LANGUAGES CXX)

include(CTest)
enable_testing()

set(CONSTRAINT_REASONER_TYPES Z3 MathSAT SeMiTONE)
set(CONSTRAINT_REASONER_TYPE SeMiTONE CACHE STRING "Constraint reasoner type")
set_property(CACHE CONSTRAINT_REASONER_TYPE PROPERTY STRINGS ${CONSTRAINT_REASONER_TYPES})
list(FIND CONSTRAINT_REASONER_TYPES ${CONSTRAINT_REASONER_TYPE} CONSTRAINT_REASONER_TYPE_INDEX)
if(${CONSTRAINT_REASONER_TYPE_INDEX} EQUAL -1)
    message(FATAL_ERROR "Invalid constraint reasoner type ${CONSTRAINT_REASONER_TYPE}")
endif()

set(TEMPORAL_NETWORK_TYPES LRA DL)
set(TEMPORAL_NETWORK_TYPE LRA CACHE STRING "Temporal network type")
set_property(CACHE TEMPORAL_NETWORK_TYPE PROPERTY STRINGS ${TEMPORAL_NETWORK_TYPES})
list(FIND TEMPORAL_NETWORK_TYPES ${TEMPORAL_NETWORK_TYPE} TEMPORAL_NETWORK_TYPE_INDEX)
if(${TEMPORAL_NETWORK_TYPE_INDEX} EQUAL -1)
    message(FATAL_ERROR "Invalid temporal network type ${TEMPORAL_NETWORK_TYPE}")
endif()

set(HEURISTIC_TYPES h_max h_add)
set(HEURISTIC_TYPE h_max CACHE STRING "Heuristic type")
set_property(CACHE HEURISTIC_TYPE PROPERTY STRINGS ${HEURISTIC_TYPES})
list(FIND HEURISTIC_TYPES ${HEURISTIC_TYPE} HEURISTIC_TYPE_INDEX)
if(${HEURISTIC_TYPE_INDEX} EQUAL -1)
    message(FATAL_ERROR "Invalid heuristic type ${HEURISTIC_TYPE}")
endif()

set(BUILD_LISTENERS ON CACHE BOOL "Builds listeners" FORCE)

option(CHECK_INCONSISTENCIES "Check inconsistencies at each step" OFF)
option(ENABLE_API "Enable the oRatio API" OFF)
option(BUILD_EXECUTABLE "Build the oRatio executable" ON)
option(BUILD_ORATIO_SERVER "Build the oRatio server" OFF)

if(BUILD_ORATIO_SERVER)
    set(BUILD_EXECUTABLE ON CACHE BOOL "Build the oRatio executable" FORCE)
    set(ENABLE_API ON CACHE BOOL "Enable the oRatio API" FORCE)
    set(COMPUTE_NAMES ON CACHE BOOL "Compute RiDDLe names" FORCE)
endif()

message(STATUS "Constraint reasoner type: ${CONSTRAINT_REASONER_TYPE}")
message(STATUS "Temporal network type: ${TEMPORAL_NETWORK_TYPE}")
message(STATUS "Heuristic type: ${HEURISTIC_TYPE}")
message(STATUS "Check inconsistencies at each step: ${CHECK_INCONSISTENCIES}")
message(STATUS "Enable the oRatio API: ${ENABLE_API}")
message(STATUS "Build the oRatio executable: ${BUILD_EXECUTABLE}")
message(STATUS "Build the oRatio server: ${BUILD_ORATIO_SERVER}")

add_subdirectory(extern/riddle)

add_library(oRatioLib src/graph.cpp src/types.cpp)
target_compile_features(oRatioLib PUBLIC cxx_std_17)
add_dependencies(oRatioLib RiDDLe)
target_link_libraries(oRatioLib PUBLIC RiDDLe)
target_include_directories(oRatioLib PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>)

if(${CONSTRAINT_REASONER_TYPE} STREQUAL "Z3")
    include(FetchContent)
    FetchContent_Declare(z3 GIT_REPOSITORY https://github.com/Z3Prover/z3 GIT_TAG z3-4.13.0)
    FetchContent_MakeAvailable(z3)
    set(z3_SOURCE_DIR_VALUE "${z3_SOURCE_DIR}")
    configure_file(${CMAKE_CURRENT_SOURCE_DIR}/CTestCustom.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/CTestCustom.cmake @ONLY)

    target_include_directories(oRatioLib PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include/z3> ${z3_SOURCE_DIR}/src/api)
    target_sources(oRatioLib PRIVATE src/z3/z3solver.cpp src/z3/z3flaws.cpp src/z3/z3types.cpp)
    target_link_libraries(oRatioLib PUBLIC libz3)
    target_compile_definitions(oRatioLib PUBLIC Z3)
elseif(${CONSTRAINT_REASONER_TYPE} STREQUAL "MathSAT")
    target_include_directories(oRatioLib PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include/msat> ${CMAKE_CURRENT_SOURCE_DIR}/../mathsat-5.6.11-linux-x86_64/include)
    target_link_directories(oRatioLib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../mathsat-5.6.11-linux-x86_64/lib)
    target_link_libraries(oRatioLib PUBLIC mathsat)
    target_compile_definitions(oRatioLib PUBLIC MathSAT)
elseif(${CONSTRAINT_REASONER_TYPE} STREQUAL "SeMiTONE")
    target_include_directories(oRatioLib PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include/st>)
    target_sources(oRatioLib PRIVATE src/st/stsolver.cpp src/st/stflaws.cpp src/st/sttypes.cpp)
    add_subdirectory(extern/semitone)
    add_dependencies(oRatioLib SeMiTONE)
    target_link_libraries(oRatioLib PUBLIC SeMiTONE)
    target_compile_definitions(oRatioLib PUBLIC SEMITONE)
endif()

if(${TEMPORAL_NETWORK_TYPE} STREQUAL "LRA")
    set(INIT_STRING "predicate Impulse(real at) { at >= origin; at <= horizon; } predicate Interval(real start, real end, real duration) { start >= origin; duration == end - start; duration >= 0.0; end <= horizon; } real origin, horizon; origin >= 0.0; origin <= horizon;" CACHE STRING "Initialization string")
    target_compile_definitions(oRatioLib PRIVATE LRA_TN)
elseif(${TEMPORAL_NETWORK_TYPE} STREQUAL "DL")
    set(INIT_STRING "predicate Impulse(time at) { at >= origin; at <= horizon; } predicate Interval(time start, time end) { start >= origin; start <= end; end <= horizon; } time origin, horizon; origin >= 0.0; origin <= horizon;" CACHE STRING "Initialization string")
    target_compile_definitions(oRatioLib PRIVATE DL_TN)
endif()
configure_file(include/init.hpp.in init.hpp @ONLY)
target_include_directories(oRatioLib PRIVATE $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>)

if(${HEURISTIC_TYPE} STREQUAL "h_max")
    target_compile_definitions(oRatioLib PRIVATE H_MAX)
elseif(${HEURISTIC_TYPE} STREQUAL "h_add")
    target_compile_definitions(oRatioLib PRIVATE H_ADD)
endif()

if(CHECK_INCONSISTENCIES)
    target_compile_definitions(oRatioLib PRIVATE CHECK_INCONSISTENCIES)
endif()

if(ENABLE_API)
    target_compile_definitions(oRatioLib PRIVATE ENABLE_API)
endif()

if (BUILD_EXECUTABLE)
    add_executable(oRatio src/exec/main.cpp)
    add_dependencies(oRatio oRatioLib)
    target_link_libraries(oRatio PUBLIC oRatioLib)

    if(BUILD_ORATIO_SERVER)
        target_compile_definitions(oRatio PRIVATE BUILD_SERVER)
        target_sources(oRatio PRIVATE src/server/solver_api.cpp src/server/solver_server.cpp)
        target_include_directories(oRatio PRIVATE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include/server>)
        if(NOT TARGET ratioNet)
            add_subdirectory(extern/rationet)
            add_dependencies(oRatio ratioNet)
        endif()
        target_link_libraries(oRatio PUBLIC ratioNet)

        add_custom_target(npm_core_install ALL COMMAND npm install WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/gui/core COMMENT "Installing the TypeScript framework" VERBATIM)
        add_custom_target(npm_core_build ALL COMMAND npm run build WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/gui/core COMMENT "Building the TypeScript framework" VERBATIM DEPENDS npm_core_install)
        add_custom_target(npm_lib_install ALL COMMAND npm install WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/gui/lib COMMENT "Installing the oRatio WebLib" VERBATIM DEPENDS npm_core_build)
        add_custom_target(npm_lib_build ALL COMMAND npm run build WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/gui/lib COMMENT "Building the oRatio WebLib" VERBATIM DEPENDS npm_lib_install)
        add_custom_target(npm_app_install ALL COMMAND npm install WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/gui/app COMMENT "Installing the oRatio WebApp" VERBATIM DEPENDS npm_lib_build)
        add_custom_target(npm_app_build ALL COMMAND npm run build WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/gui/app COMMENT "Building the oRatio WebApp" VERBATIM DEPENDS npm_app_install)
    endif()

    setup_sanitizers(oRatio)
endif()

setup_sanitizers(oRatioLib)

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()

set(CPACK_PROJECT_NAME ${PROJECT_NAME})
set(CPACK_PROJECT_VERSION ${PROJECT_VERSION})
include(CPack)
